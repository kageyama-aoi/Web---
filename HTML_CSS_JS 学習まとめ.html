12) JS：テーマの優先順位（ミニ版・保存あり）

プレビュー（保存あり）

選択は localStorage に保存され、再読込で復元されます。
(function(){
  const wrap = document.getElementById("mini-theme-persist");
  const key = wrap.dataset.key || "mini-theme";
  const saved = localStorage.getItem(key);
  if(saved) wrap.setAttribute("data-theme", saved);
  wrap.querySelectorAll("[data-mini-theme]").forEach(btn => {
    btn.addEventListener("click", () => {
      const mode = btn.dataset.miniTheme;
      wrap.setAttribute("data-theme", mode);
      localStorage.setItem(key, mode);
      wrap.querySelectorAll("[data-mini-theme].active").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
    });
  });
})();
保存キーを data-key で変えれば、他のプレビューと衝突しません。

14) details/summary アコーディオン＋スムースアニメ（アクセシブル）

プレビュー（クリックで開閉・アニメーション）

アコーディオン1：概要

HTMLの <details><summary> を使った、アクセシブルなアコーディオンです。キーボード（Enter/Space）でも開閉できます。

アコーディオン2：仕様
  • ネイティブの開閉UI（スクリーンリーダー対応）
  • Web Animations API で高さをアニメーション
  • 開閉途中の連打に対するキャンセル処理
/* HTML */
<details class="acc js-acc">
  <summary>見出し</summary>
  <div class="acc-content">中身...</div>
</details>

/* CSS（見た目） */
.acc{border:1px solid #ddd; border-radius:8px; padding:.4rem .6rem; background:#fff; margin:.5rem 0}
.acc > summary{cursor:pointer; list-style:none; font-weight:600}
.acc > summary::-webkit-details-marker{display:none}
.acc .acc-content{overflow:clip} /* WAAPIでheightをアニメするのでクリップ */

/* JS（アニメーション） */
(function(){
  const EASING = 'cubic-bezier(.2,.6,.2,1)';
  const DURATION = 200;
  function setup(details){
    let anim = null;
    const content = details.querySelector('.acc-content');
    details.addEventListener('click', (e)=>{
      if(e.target.tagName.toLowerCase() !== 'summary') return;
      e.preventDefault();
      if(!details.open){ open(); } else { close(); }
    });
    function close(){
      const start = content.offsetHeight, end = 0;
      anim?.cancel();
      anim = content.animate([{height:start+'px'},{height:end+'px'}], {duration:DURATION, easing:EASING});
      anim.onfinish = ()=>{ details.open = false; content.style.height=''; };
    }
    function open(){
      details.open = true;
      requestAnimationFrame(()=>{
        const start = 0, end = content.offsetHeight;
        content.style.height = start+'px';
        anim?.cancel();
        anim = content.animate([{height:start+'px'},{height:end+'px'}], {duration:DURATION, easing:EASING});
        anim.onfinish = ()=>{ content.style.height=''; };
      });
    }
  }
  document.querySelectorAll('.js-acc').forEach(setup);
})();
details/summaryはデフォルトでアクセシビリティに優れます。ここではWeb Animations APIで“高さ”を補間して、自然な開閉にしています。

16) タブ UI（ARIA対応・ボタン＋tabpanel）

プレビュー（クリックでタブ切替）

ARIAロール（tablist/tab/tabpanel）で構造を示し、選択中タブに aria-selected="true" を付けます。

  • ボタンに aria-controls で対象パネルIDを紐づけ。
  • 選択切替で hidden の出し入れ。

キーボード操作（←→）を付けるとより親切です。今回はクリック切替の最小実装。

<div class="tabs" data-tabs>
  <div class="tablist" role="tablist">
    <button class="tab" role="tab" aria-selected="true" aria-controls="p1" id="t1">概要</button>
    <button class="tab" role="tab" aria-selected="false" aria-controls="p2" id="t2">使い方</button>
  </div>
  <div class="tabpanel" role="tabpanel" id="p1" aria-labelledby="t1">...</div>
  <div class="tabpanel" role="tabpanel" id="p2" aria-labelledby="t2" hidden>...</div>
</div>

<script>
document.addEventListener("click", (e)=>{
  const btn = e.target.closest(".tabs [role='tab']");
  if(!btn) return;
  const wrap = btn.closest(".tabs");
  wrap.querySelectorAll("[role='tab']").forEach(t => t.setAttribute("aria-selected", String(t===btn)));
  wrap.querySelectorAll("[role='tabpanel']").forEach(p => p.hidden = true);
  const id = btn.getAttribute("aria-controls");
  wrap.querySelector("#"+id).hidden = false;
});
</script>

17) トースト通知（自動クローズ・ARIAライブ）

プレビュー（通知を表示）

<div class="toast-area" aria-live="polite" aria-atomic="true"></div>
<script>
function spawnToast(area, message, type="info"){
  const el = document.createElement("div");
  el.className = "toast " + type;
  el.role = "status";
  el.textContent = message;
  area.appendChild(el);
  setTimeout(() => el.remove(), 2500);
}
document.addEventListener("click", (e)=>{
  const btn = e.target.closest("[data-toast]");
  if(!btn) return;
  const area = btn.closest("[data-toast-area]").querySelector(".toast-area");
  spawnToast(area, btn.dataset.toast.toUpperCase()+" Toast", btn.dataset.toast);
});
</script>
